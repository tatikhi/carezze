    function pluralDaysWord(n) {
      const abs = Math.abs(Math.trunc(Number(n) || 0));
      const mod100 = abs % 100;
      if (mod100 >= 11 && mod100 <= 19) return 'дней';
      const mod10 = abs % 10;
      if (mod10 === 1) return 'день';
      if (mod10 >= 2 && mod10 <= 4) return 'дня';
      return 'дней';
    }

    function pluralDays(n) {
      return `${n} ${pluralDaysWord(n)}`;
    }

    function setupKeyboardLift() {
      const updateViewportOffset = () => {
        if (!window.visualViewport) return;
        const vv = window.visualViewport;
        const offset = Math.max(0, window.innerHeight - vv.height - vv.offsetTop);
        document.documentElement.style.setProperty('--kb-offset', `${Math.round(offset)}px`);
      };

      if (window.visualViewport) {
        window.visualViewport.addEventListener('resize', updateViewportOffset);
        window.visualViewport.addEventListener('scroll', updateViewportOffset);
      }

      document.addEventListener('focusin', (e) => {
        const el = e.target;
        if (!el || !['INPUT', 'TEXTAREA'].includes(el.tagName)) return;
        if (el.type === 'date') return;

        document.body.classList.add('keyboard-open');
        setTimeout(() => {
          updateViewportOffset();
          try { el.scrollIntoView({ behavior: 'smooth', block: 'center' }); } catch (err) {}
        }, 280);
      });

      document.addEventListener('focusout', () => {
        setTimeout(() => {
          const active = document.activeElement;
          if (!active || !['INPUT', 'TEXTAREA'].includes(active.tagName)) {
            document.body.classList.remove('keyboard-open');
            document.documentElement.style.setProperty('--kb-offset', '0px');
          }
        }, 120);
      });
    }

    function utf8ByteLength(str) {
      try {
        return new TextEncoder().encode(str).length;
      } catch (e) {
        return unescape(encodeURIComponent(str)).length;
      }
    }

    function formatLocalISO(dateObj) {
      const y = dateObj.getFullYear();
      const m = String(dateObj.getMonth() + 1).padStart(2, '0');
      const d = String(dateObj.getDate()).padStart(2, '0');
      return `${y}-${m}-${d}`;
    }

    function parseLocalISO(isoStr) {
      const [y, m, d] = isoStr.split('-').map(Number);
      return new Date(y, m - 1, d, 12, 0, 0);
    }

    function getTodayISO(offsetDays = 0) {
      const d = new Date();
      d.setHours(12, 0, 0, 0);
      d.setDate(d.getDate() + offsetDays);
      return formatLocalISO(d);
    }

    function shiftISOByDays(isoStr, deltaDays) {
      const d = parseLocalISO(isoStr);
      d.setDate(d.getDate() + deltaDays);
      return formatLocalISO(d);
    }

    function formatHumanDate(isoStr) {
      const today = getTodayISO(0);
      const yest = getTodayISO(-1);
      const tom = getTodayISO(1);
      const d = parseLocalISO(isoStr);
      const longDay = d.toLocaleDateString('ru-RU', { day: 'numeric', month: 'long' });
      if (isoStr === today) return `${longDay} · Сегодня`;
      if (isoStr === yest) return `${longDay} · Вчера`;
      if (isoStr === tom) return `${longDay} · Завтра`;
      return longDay;
    }

    function calculateAgeFromBirthDate(birthDateStr) {
      if (!birthDateStr || !/^\d{4}-\d{2}-\d{2}$/.test(birthDateStr)) return '';
      const [by, bm, bd] = birthDateStr.split('-').map(Number);
      const now = new Date();
      let age = now.getFullYear() - by;
      const monthDiff = (now.getMonth() + 1) - bm;
      if (monthDiff < 0 || (monthDiff === 0 && now.getDate() < bd)) {
        age -= 1;
      }
      if (age < 10 || age > 100) return '';
      return age;
    }

    function estimateBirthDateFromAge(ageVal) {
      const ageNum = Number(ageVal);
      if (!ageNum || ageNum < 10 || ageNum > 100) return '';
      const now = new Date();
      const approxYear = now.getFullYear() - ageNum;
      const m = String(now.getMonth() + 1).padStart(2, '0');
      const d = String(now.getDate()).padStart(2, '0');
      return `${approxYear}-${m}-${d}`;
    }

    function smartTrimText(text, maxLen) {
      const str = String(text || '').trim();
      if (!str || str.length <= maxLen) return str;
      if (maxLen <= 1) return '';
      const cut = str.slice(0, maxLen - 1);
      const lastSentence = Math.max(cut.lastIndexOf('. '), cut.lastIndexOf('! '), cut.lastIndexOf('? '), cut.lastIndexOf('…'));
      if (lastSentence >= Math.floor(maxLen * 0.55)) {
        return cut.slice(0, lastSentence + 1).trim();
      }
      const lastSpace = cut.lastIndexOf(' ');
      if (lastSpace >= Math.floor(maxLen * 0.5)) {
        return cut.slice(0, lastSpace).trim() + '…';
      }
      return cut.trim() + '…';
    }

    function getAiAnalysisTextLength(ai) {

      const stepsLen = Array.isArray(ai.steps)

        + stepsLen
        + tipsLen;
    }

    function compactEntryForCloud(entry, aiMaxScale = 1) {
      const compactEmotions = {};
      if (entry.emotionLevels && typeof entry.emotionLevels === 'object') {
        Object.entries(entry.emotionLevels).forEach(([k, v]) => {
          const num = Number(v);
          if (Number.isFinite(num) && num > 0) {
            compactEmotions[k] = Math.max(0, Math.min(5, Math.round(num)));
          }
        });
      }
      const soloCnt = normalizeSoloCount(entry);
      const clone = {
        date: entry.date,
        scaleMax: 5,
        updatedAt: entry.updatedAt || Date.now(),
        isPeriod: Boolean(entry.isPeriod),
        isOvulation: Boolean(entry.isOvulation),
        intimacy: Boolean(entry.intimacy),
        soloCount: soloCnt,
        soloReset: soloCnt > 0,
        discharge: entry.discharge || '',
        mood: entry.mood,
        energy: entry.energy,
        pain: entry.pain,
        libido: entry.libido,
        stress: Math.max(0, Math.min(5, Math.round(Number(entry.stress) || 0))),
        emotions: Object.keys(compactEmotions),
        emotionLevels: compactEmotions,
        contexts: entry.contexts || [],
        helped: Array.isArray(entry.helped) ? entry.helped.slice(0, 20) : []
      };
      if (typeof entry.wristTempDelta === 'number' && Number.isFinite(entry.wristTempDelta)) {
        clone.wristTempDelta = entry.wristTempDelta;
      }
      if (entry.aiAnalysis && aiMaxScale > 0) {
        if (aiMaxScale >= 1) {
          clone.aiAnalysis = {
            physiology: String(entry.aiAnalysis.physiology || ''),
            patterns: String(entry.aiAnalysis.patterns || ''),
            focus: String(entry.aiAnalysis.focus || ''),
            steps: Array.isArray(entry.aiAnalysis.steps) ? entry.aiAnalysis.steps.slice(0, 4).map(s => String(s || '')) : [],
            tips: Array.isArray(entry.aiAnalysis.tips) ? entry.aiAnalysis.tips.slice(0, 4).map(t => ({
              title: String(t?.title || ''),
              sub: String(t?.sub || '')
            })) : []
          };
        } else {
          const pLen = Math.round(620 * aiMaxScale);
          const fLen = Math.round(480 * aiMaxScale);
          const stepLen = Math.round(240 * aiMaxScale);
          const titleLen = Math.round(120 * aiMaxScale);
          const subLen = Math.round(220 * aiMaxScale);
          clone.aiAnalysis = {
            physiology: smartTrimText(entry.aiAnalysis.physiology || '', pLen),
            patterns: smartTrimText(entry.aiAnalysis.patterns || '', pLen),
            focus: smartTrimText(entry.aiAnalysis.focus || '', fLen),
            steps: (entry.aiAnalysis.steps || []).slice(0, 4).map(s => smartTrimText(s, stepLen)),
            tips: (entry.aiAnalysis.tips || []).slice(0, 4).map(t => ({
              title: smartTrimText(t?.title || '', titleLen),
              sub: smartTrimText(t?.sub || '', subLen)
            }))
          };
        }
      }
      return clone;
    }

    function serializeDaySafelyForCloud(entry) {
      const scales = [1, 0.85, 0.7, 0.55, 0.4, 0];

          return serialized;
        }
      }
      return serialized;
    }

    function mergeEntryRecords(cloudEntry, localEntry) {

      if (!localEntry) return cloudEntry;
      const cTime = Number(cloudEntry.updatedAt) || 0;
      const lTime = Number(localEntry.updatedAt) || 0;
      const newer = lTime >= cTime ? localEntry : cloudEntry;
      const older = lTime >= cTime ? cloudEntry : localEntry;

      const merged = { ...older, ...newer };

      const newerLevels = newer.emotionLevels && typeof newer.emotionLevels === 'object' ? newer.emotionLevels : {};
      const olderLevels = older.emotionLevels && typeof older.emotionLevels === 'object' ? older.emotionLevels : {};
      const newerHasEmos = Object.values(newerLevels).some(v => Number(v) > 0);
      const olderHasEmos = Object.values(olderLevels).some(v => Number(v) > 0);

      if (newerHasEmos) {
        merged.emotionLevels = { ...createDefaultEmotionLevels(), ...newerLevels };
      } else if (olderHasEmos) {
        merged.emotionLevels = { ...createDefaultEmotionLevels(), ...olderLevels };
      } else {
        merged.emotionLevels = buildLoadedEmotionLevels(newer);
      }

      const soloCnt = Math.max(normalizeSoloCount(newer), lTime >= cTime ? 0 : normalizeSoloCount(older));
      merged.soloCount = normalizeSoloCount(newer);
      merged.soloReset = merged.soloCount > 0;
      if (merged.soloCount === 0 && soloCnt > 0 && lTime === cTime) {
        merged.soloCount = soloCnt;
        merged.soloReset = true;
      }

      if ((merged.wristTempDelta === undefined || merged.wristTempDelta === null) && typeof older.wristTempDelta === 'number') {
        merged.wristTempDelta = older.wristTempDelta;
      }
      if (!merged.aiAnalysis && older.aiAnalysis) {
        merged.aiAnalysis = older.aiAnalysis;
      } else if (merged.aiAnalysis && older.aiAnalysis && Math.abs(lTime - cTime) < 2000) {
        if (getAiAnalysisTextLength(older.aiAnalysis) > getAiAnalysisTextLength(merged.aiAnalysis)) {
          merged.aiAnalysis = older.aiAnalysis;
        }
      }
      if (!Array.isArray(merged.helped)) {
        merged.helped = Array.isArray(older.helped) ? [...older.helped] : [];
      } else if (Math.abs(lTime - cTime) < 2000 && Array.isArray(older.helped) && older.helped.length) {
        merged.helped = Array.from(new Set([...merged.helped, ...older.helped]));
      }
      if (!merged.updatedAt) {
        merged.updatedAt = Math.max(cTime, lTime, Date.now());
      }
      return merged;
    }

    function serializeCustomPoolSafely(pool) {
      const clean = Array.isArray(pool) ? pool.map(x => String(x || '').trim()).filter(Boolean) : [];
      let slice = clean.slice(0, 30);
      let str = JSON.stringify(slice);
      while (slice.length > 0 && utf8ByteLength(str) > 3900) {
        slice.pop();
        str = JSON.stringify(slice);
      }
      return str;
    }

    async function syncToCloud(specificTargetKey = null) {
      if (!hasTelegramCloud()) {
        document.getElementById('cloudSyncBadge').textContent = '';
        return;
      }
      document.getElementById('cloudSyncBadge').textContent = 'Cloud';

      const tasks = [];
      const entriesList = Object.entries(state.entries);

      if (specificTargetKey !== '__meta_only__') {
        for (const [iso, entry] of entriesList) {
          if (!specificTargetKey || iso === specificTargetKey || iso.startsWith(specificTargetKey)) {
            const serialized = serializeDaySafelyForCloud(entry);
            tasks.push(cloudSet(`carezze2_d_${iso}`, serialized));
          }
        }
      }

      const customPool = Array.isArray(state.customHelpedPool) ? state.customHelpedPool.slice(0, 30) : [];
      tasks.push(cloudSet('carezze2_custom_tips', serializeCustomPoolSafely(customPool)));

      let metaPool = customPool.slice(0, 20);
      let metaPayload = JSON.stringify({
        onboarded: state.onboarded,
        settings: state.settings,
        painkillers: state.painkillers.slice(0, 12),
        customHelpedPool: metaPool
      });
      while (metaPool.length > 0 && utf8ByteLength(metaPayload) > 3950) {
        metaPool.pop();
        metaPayload = JSON.stringify({
          onboarded: state.onboarded,
          settings: state.settings,
          painkillers: state.painkillers.slice(0, 12),
          customHelpedPool: metaPool
        });
      }
      tasks.push(cloudSet('carezze2_meta', metaPayload));

      await Promise.all(tasks);
    }

    function toFiveScale(val) {
      const n = Number(val);
      if (!Number.isFinite(n) || n <= 0) return 0;
      return Math.max(0, Math.min(5, Math.round(n / 2)));
    }

    function normalizeAllLoadedEntries() {
      Object.keys(state.entries || {}).forEach(iso => {
        const entry = state.entries[iso];
        if (!entry) return;
        entry.date = entry.date || iso;
        const legacyStressRaw = entry.emotionLevels?.['Стресс'];
        const legacyContextStress = Array.isArray(entry.contexts) && entry.contexts.includes('Стресс') ? 3 : 0;
        const isLegacy10 = entry.scaleMax !== 5 && (
          Number(entry.energy) > 5 ||
          Number(entry.pain) > 5 ||
          Number(entry.libido) > 5 ||
          Number(entry.stress) > 5 ||
          Object.values(entry.emotionLevels || {}).some(v => Number(v) > 5)
        );
        if (isLegacy10) {
          entry.energy = toFiveScale(entry.energy);
          entry.pain = toFiveScale(entry.pain);
          entry.libido = toFiveScale(entry.libido);
          const rawS = entry.stress !== undefined ? entry.stress : (legacyStressRaw !== undefined ? legacyStressRaw : 0);
          entry.stress = toFiveScale(rawS) || legacyContextStress;
          if (typeof entry.mood === 'number') entry.mood = toFiveScale(entry.mood);
          if (entry.emotionLevels && typeof entry.emotionLevels === 'object') {
            Object.keys(entry.emotionLevels).forEach(k => {
              entry.emotionLevels[k] = toFiveScale(entry.emotionLevels[k]);
            });
          }
        } else {
          entry.energy = Math.max(0, Math.min(5, Math.round(Number(entry.energy) || 0)));
          entry.pain = Math.max(0, Math.min(5, Math.round(Number(entry.pain) || 0)));
          entry.libido = Math.max(0, Math.min(5, Math.round(Number(entry.libido) || 0)));
          const rawS = entry.stress !== undefined ? entry.stress : (legacyStressRaw !== undefined ? legacyStressRaw : legacyContextStress);
          entry.stress = Math.max(0, Math.min(5, Math.round(Number(rawS) || 0)));
        }
        entry.scaleMax = 5;
        entry.emotionLevels = buildLoadedEmotionLevels(entry);
        entry.emotions = getActiveEmotionsFromLevels(entry.emotionLevels);
        entry.soloCount = normalizeSoloCount(entry);
        entry.soloReset = entry.soloCount > 0;
      });
    }

    async function syncFromCloudBackground() {
      if (!hasTelegramCloud() || isCloudSyncing) return;
      isCloudSyncing = true;
      try {
        await pendingCloudWritePromise;
        if (Date.now() - lastLocalWriteTimestamp < 4000) {
          return;
        }

        document.getElementById('cloudSyncBadge').textContent = 'Cloud';
        const metaRaw = (await cloudGet('carezze2_meta'))
          || (await cloudGet('carezze1_meta'))
          || (await cloudGet('ritm9_meta'))
          || (await cloudGet('ritm8_meta'));

        if (metaRaw) {
          const meta = JSON.parse(metaRaw);
          state.onboarded = Boolean(meta.onboarded) || state.onboarded;
          state.settings = { ...state.settings, ...(meta.settings || {}) };
          if (!THEMES[state.settings.palette]) {
            state.settings.palette = 'peach_sage';
          }
          ensureProfileStructure();
          if (Array.isArray(meta.painkillers) && meta.painkillers.length) {
            state.painkillers = meta.painkillers;
          }
          if (Array.isArray(meta.customHelpedPool)) {
            state.customHelpedPool = Array.from(new Set([...state.customHelpedPool, ...meta.customHelpedPool]));
          }
        }

        const customTipsRaw = await cloudGet('carezze2_custom_tips');
        if (customTipsRaw) {
          try {
            const parsedTips = JSON.parse(customTipsRaw);
            if (Array.isArray(parsedTips)) {
              state.customHelpedPool = Array.from(new Set([...state.customHelpedPool, ...parsedTips]));
            }
          } catch (e) {}
        }

        const keys = await cloudGetKeys();
        const cloudEntriesMap = {};

        const monthKeys = keys.filter(k =>
          k.startsWith('carezze2_m_') || k.startsWith('carezze1_m_') || k.startsWith('ritm9_m_') || k.startsWith('ritm8_m_')
        );
        for (const mk of monthKeys) {
          const mRaw = await cloudGet(mk);
          if (mRaw) {
            const mEntries = JSON.parse(mRaw);
            Object.entries(mEntries).forEach(([iso, entry]) => {
              cloudEntriesMap[iso] = mergeEntryRecords(cloudEntriesMap[iso], entry);
            });
          }
        }

        const dayKeys = keys.filter(k => k.startsWith('carezze2_d_'));
        for (const dk of dayKeys) {
          const dRaw = await cloudGet(dk);
          if (dRaw) {
            const dEntry = JSON.parse(dRaw);
            const iso = dEntry.date || dk.replace('carezze2_d_', '');
            cloudEntriesMap[iso] = mergeEntryRecords(cloudEntriesMap[iso], dEntry);
          }
        }

        const allDates = new Set([...Object.keys(cloudEntriesMap), ...Object.keys(state.entries)]);
        allDates.forEach(iso => {
          state.entries[iso] = mergeEntryRecords(cloudEntriesMap[iso], state.entries[iso]);
        });

        normalizeAllLoadedEntries();

        if (importedTempToday !== null) {
          applyIncomingTempToToday(importedTempToday, false);
        }

        ensureOnboardedFlagIfDataExists();
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

        if (importedTempToday !== null) {
          await syncToCloud(getTodayISO(0));
        }

        applyThemeUI();
        syncSettingsInputs();
        if (state.onboarded) {
          document.getElementById('onboardingScreen').classList.remove('active');
        }
        loadDayForm();
        renderAnalytics();
      } catch (e) {
      } finally {
        isCloudSyncing = false;
      }
    }

    async function callGeminiAPI(apiKey, promptText, expectJson = true) {
      const isExpressKey = apiKey.startsWith('AQ.');
      const models = ['gemini-3.8-flash', 'gemini-flash-latest', 'gemini-3.5-flash-lite'];
      let lastError = 'Ошибка запроса';

      const payload = {
        contents: [{ role: 'user', parts: [{ text: promptText }] }],
        generationConfig: {
          temperature: 0.55,
          maxOutputTokens: 4096,
          ...(expectJson ? { responseMimeType: 'application/json' } : {})
        }
      };

      for (const model of models) {
        const endpoints = isExpressKey
          ? [
              `https://aiplatform.googleapis.com/v1/publishers/google/models/${model}:generateContent?key=${encodeURIComponent(apiKey)}`,
              `https://aiplatform.googleapis.com/v1beta1/publishers/google/models/${model}:generateContent?key=${encodeURIComponent(apiKey)}`,
              `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${encodeURIComponent(apiKey)}`
            ]
          : [
              `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${encodeURIComponent(apiKey)}`
            ];

        for (const url of endpoints) {
          try {
            const res = await fetch(url, {
              method: 'POST',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify(payload)
            });

            const data = await res.json();
            if (!res.ok) {
              const errMsg = data?.error?.message || `HTTP ${res.status}`;
              lastError = errMsg;
              if (res.status === 404 || res.status === 400 || res.status === 403) continue;
              throw new Error(errMsg);
            }

            const candidate = data?.candidates?.[0];
            const parts = Array.isArray(candidate?.content?.parts) ? candidate.content.parts : [];
            const rawText = parts
              .filter(p => !p.thought && typeof p.text === 'string')
              .map(p => p.text)
              .join('') || parts[0]?.text || '';
            if (!expectJson) return rawText.trim();
            const cleaned = rawText.replace(/^```json\s*/i, '').replace(/```\s*$/i, '').trim();
            return JSON.parse(cleaned);
          } catch (e) {
            lastError = e.message || String(e);
          }
        }
      }
      throw new Error(lastError);
    }

    function attachHorizontalSlider(el, getVal, setVal, renderFn, onCommit = null) {
      const min = Number(el.dataset.min);
      const max = Number(el.dataset.max);
      let startX = 0;
      let startY = 0;
      let gestureMode = null;
      let cachedRect = null;
      let lastClientX = 0;

      const calcFromClientX = (clientX) => {
        const rect = cachedRect || el.getBoundingClientRect();
        let ratio = (clientX - rect.left) / Math.max(1, rect.width);
        ratio = Math.max(0, Math.min(1, ratio));
        const val = Math.round(min + ratio * (max - min));
        return { val, ratio };
      };

      const applyPoint = (clientX, smoothFollow = false) => {
        const { val, ratio } = calcFromClientX(clientX);
        const prev = getVal();
        if (prev !== val) {
          setVal(val);
          hapticImpact('light');
          hapticSelect();
        }
        renderFn(val, smoothFollow ? ratio : null);
      };

      el.addEventListener('touchstart', (e) => {
        if (e.touches.length !== 1) return;
        const t = e.touches[0];
        startX = t.clientX;
        startY = t.clientY;
        lastClientX = t.clientX;
        gestureMode = null;
        cachedRect = el.getBoundingClientRect();
      }, { passive: true });

      el.addEventListener('touchmove', (e) => {
        if (e.touches.length !== 1) return;
        const t = e.touches[0];
        lastClientX = t.clientX;
        const dx = Math.abs(t.clientX - startX);
        const dy = Math.abs(t.clientY - startY);

        if (!gestureMode) {
          if (dx < 4 && dy < 4) return;
          if (dx >= dy) {
            gestureMode = 'horizontal';
            el.classList.add('is-dragging');
          } else {
            gestureMode = 'vertical';
            return;
          }
        }

        if (gestureMode === 'horizontal') {
          if (e.cancelable) e.preventDefault();
          applyPoint(t.clientX, true);
        }
      }, { passive: false });

      const finishTouch = () => {
        el.classList.remove('is-dragging');
        if (gestureMode === 'horizontal') {
          renderFn(getVal(), null);
          if (onCommit) onCommit();
        } else if (gestureMode === null) {
          applyPoint(lastClientX, false);
          if (onCommit) onCommit();
        }
        gestureMode = null;
        cachedRect = null;
      };

      el.addEventListener('touchend', finishTouch, { passive: true });
      el.addEventListener('touchcancel', () => {
        el.classList.remove('is-dragging');
        if (gestureMode === 'horizontal') {
          renderFn(getVal(), null);
          if (onCommit) onCommit();
        }
        gestureMode = null;
        cachedRect = null;
      }, { passive: true });

      let mouseActive = false;
      el.addEventListener('mousedown', (e) => {
        if (e.button !== 0) return;
        mouseActive = true;
        cachedRect = el.getBoundingClientRect();
        el.classList.add('is-dragging');
        applyPoint(e.clientX, true);
      });

      window.addEventListener('mousemove', (e) => {
        if (!mouseActive) return;
        applyPoint(e.clientX, true);
      });

      window.addEventListener('mouseup', () => {
        if (!mouseActive) return;
        mouseActive = false;
        el.classList.remove('is-dragging');
        renderFn(getVal(), null);
        cachedRect = null;
        if (onCommit) onCommit();
      });
    }

    function getCycleInfo(dateISO) {
      const start = parseLocalISO(state.settings.lastPeriodStart);
      const target = parseLocalISO(dateISO);
      const today = parseLocalISO(getTodayISO(0));
      const diffDays = Math.round((target - start) / (1000 * 60 * 60 * 24));
      const len = Number(state.settings.cycleLength) || 28;
      const pLen = Number(state.settings.periodLength) || 5;

      const savedEntry = state.entries[dateISO];
      const isCycleStartDay = dateISO === state.settings.lastPeriodStart;
      const isMarkedPeriod = (dateISO === selectedDate)
        ? tempIsPeriod
        : Boolean(savedEntry?.isPeriod || isCycleStartDay);
      const isMarkedOvulation = (dateISO === selectedDate)
        ? tempIsOvulation
        : Boolean(savedEntry?.isOvulation);

      const contraception = state.settings?.profile?.contraception || 'Нет';
      const isSuppressedOvulation = contraception === 'КОК' || contraception === 'Кольцо / пластырь';
      const isHormonalIUD = contraception === 'Гормональная ВМС';

      const ovulationDay = Math.max(pLen + 2, len - 14);
      const isMidCycleWindow = (cycleD) => !isSuppressedOvulation && cycleD >= Math.max(8, ovulationDay - 4) && cycleD <= ovulationDay + 3;

      let isDelay = false;
      let delayDays = 0;
      let cycleDay = 1;

      const hasExplicitPeriodOnDate = Boolean(savedEntry && savedEntry.isPeriod) || (dateISO === selectedDate && tempIsPeriod);
      const hasExplicitPeriodAfterStart = Object.values(state.entries).some(
        e => e.isPeriod && e.date > state.settings.lastPeriodStart && e.date <= dateISO
      ) || (selectedDate > state.settings.lastPeriodStart && selectedDate <= dateISO && tempIsPeriod);

      if (diffDays >= 0 && !hasExplicitPeriodOnDate && !hasExplicitPeriodAfterStart) {
        cycleDay = diffDays + 1;
        if (diffDays >= len) {
          isDelay = true;
          delayDays = cycleDay - len;
        }
      } else if (diffDays >= 0) {
        cycleDay = diffDays + 1;
      } else {
        cycleDay = ((diffDays % len) + len) % len + 1;
      }

      let isPredictedPeriod = false;
      let isPredictedOvulation = false;

      const todayDiffDays = Math.round((today - start) / (1000 * 60 * 60 * 24));
      const hasOngoingDelay = todayDiffDays >= len && !Object.values(state.entries).some(
        e => e.isPeriod && e.date > state.settings.lastPeriodStart && e.date <= getTodayISO(0)
      );

      if (target > today && !isMarkedPeriod && !hasOngoingDelay) {
        const projectedDay = ((diffDays % len) + len) % len + 1;
        if (projectedDay <= pLen) {
          isPredictedPeriod = true;
        } else if (!isSuppressedOvulation && !isHormonalIUD && Math.abs(projectedDay - ovulationDay) <= 1) {
          isPredictedOvulation = true;
        }
      }

      let phase = '';
      let phaseCode = '';

      if (isMarkedPeriod) {
        phase = isSuppressedOvulation ? 'Кровотечение отмены' : 'Менструация';
        phaseCode = 'menstrual';
      } else if (isDelay) {
        phase = `Задержка · ${pluralDays(delayDays)}`;
        phaseCode = 'delay';
      } else if (isMarkedOvulation) {
        phase = 'Овуляция';
        phaseCode = 'ovulation';
      } else if (isSuppressedOvulation) {
        if (cycleDay <= len - 7) {
          phase = 'Активные дни приёма';
          phaseCode = 'follicular';
        } else {
          phase = 'Окно отмены / пауза';
          phaseCode = 'late_luteal';
        }
      } else if (cycleDay <= ovulationDay - 1) {
        phase = cycleDay <= pLen && diffDays >= 0 && diffDays < pLen
          ? 'Менструальная фаза'
          : 'Фолликулярная фаза';
        phaseCode = 'follicular';
      } else if (cycleDay <= ovulationDay + 1) {
        phase = 'Овуляторное окно';
        phaseCode = 'ovulation';
      } else if (cycleDay <= len - 6) {
        phase = 'Лютеиновая фаза';
        phaseCode = 'early_luteal';
      } else {
        phase = 'Предменструальное окно';
        phaseCode = 'late_luteal';
      }

      return {
        cycleDay,
        phase,
        phaseCode,
        len,
        pLen,
        isDelay,
        delayDays,
        isCycleStartDay,
        isMarkedPeriod,
        isPredictedPeriod,
        isPredictedOvulation,
        isSuppressedOvulation,
        isHormonalIUD,
        contraception,
        ovulationDay,
        showOvulationOption: isMidCycleWindow(cycleDay) || isMarkedOvulation
      };
    }

    function getPhaseShortHint(info) {
      if (info.isDelay) return `Задержка ${pluralDays(info.delayDays)}`;
      if (tempIsPeriod || info.phaseCode === 'menstrual') return 'Бережный режим и тепло';
      if (tempIsOvulation || info.phaseCode === 'ovulation') return 'Пик эстрогенов';
      if (info.phaseCode === 'late_luteal') return 'Скоро новый цикл';
      if (info.phaseCode === 'early_luteal') return 'Прогестероновая фаза';
      return 'Рост эстрогенов';
    }

    function findPreviousCycleStartBefore(excludeDateISO) {
      const periodDates = Object.values(state.entries)
        .filter(e => e.isPeriod && e.date < excludeDateISO)
        .map(e => e.date)
        .sort();
      if (!periodDates.length) return null;

      let blockStart = periodDates[periodDates.length - 1];
      for (let i = periodDates.length - 2; i >= 0; i--) {
        const prev = parseLocalISO(periodDates[i]);
        const curr = parseLocalISO(blockStart);
        const gap = Math.round((curr - prev) / (1000 * 60 * 60 * 24));
        if (gap <= 2) {
          blockStart = periodDates[i];
        } else {
          break;
        }
      }
      return blockStart;
    }

    function togglePeriodDay() {
      hapticImpact('medium');
      const infoBefore = getCycleInfo(selectedDate);
      const turningOn = !tempIsPeriod;

      if (turningOn) {
        tempIsPeriod = true;
        tempIsOvulation = false;
        if (!['Скудные', 'Умеренные', 'Обильные'].includes(tempDischarge)) tempDischarge = 'Умеренные';

        const prevISO = shiftISOByDays(selectedDate, -1);
        const prevWasPeriod = Boolean(state.entries[prevISO]?.isPeriod) || prevISO === state.settings.lastPeriodStart;
        const shouldStartNewCycle = (infoBefore.cycleDay >= 15 || infoBefore.isDelay)
          && !prevWasPeriod
          && selectedDate !== state.settings.lastPeriodStart;

        if (shouldStartNewCycle) {
          state.settings.prevLastPeriodStart = state.settings.lastPeriodStart;
          state.settings.lastPeriodStart = selectedDate;
          syncSettingsInputs();
        }
      } else {
        tempIsPeriod = false;
        tempDischarge = '';

        if (selectedDate === state.settings.lastPeriodStart) {
          const fallbackStart = state.settings.prevLastPeriodStart || findPreviousCycleStartBefore(selectedDate);
          if (fallbackStart && fallbackStart !== selectedDate) {
            state.settings.lastPeriodStart = fallbackStart;
            state.settings.prevLastPeriodStart = null;
            syncSettingsInputs();
          }
        }
      }

      autosaveCurrentDayDraft();
      updateHeaderAndContextControls();
      renderAdviceAndSelfCareTab();
      renderAnalytics();
    }

    function buildLoadedEmotionLevels(entry) {
      const levels = createDefaultEmotionLevels();
      if (entry?.emotionLevels && typeof entry.emotionLevels === 'object') {
        Object.entries(entry.emotionLevels).forEach(([k, v]) => {
          const normKey = k === 'Интерес' ? 'Радость' : k;
          const num = Number(v);
          if (normKey in levels && Number.isFinite(num)) {
            const scaled = (entry.scaleMax !== 5 && num > 5) ? Math.round(num / 2) : Math.round(num);
            levels[normKey] = Math.max(0, Math.min(5, scaled));
          }
        });
      }
      if (Array.isArray(entry?.emotions)) {
        const hasAnySavedVal = Object.values(levels).some(v => v > 0);
        entry.emotions.forEach(em => {
          const normKey = em === 'Интерес' ? 'Радость' : em;
          if (normKey in levels && levels[normKey] === 0 && !hasAnySavedVal) {
            levels[normKey] = 3;
          }
        });
      }

      return levels;
    }

    function loadDayForm() {
      const existing = state.entries[selectedDate];
      const isCycleStart = selectedDate === state.settings.lastPeriodStart;
      isEditingDay = true;

      if (existing) {
        const legacyStress = Number(existing.emotionLevels?.['Стресс']) || (Array.isArray(existing.contexts) && existing.contexts.includes('Стресс') ? 3 : 0);
        metrics = {
          energy: Math.max(0, Math.min(5, Math.round(Number(existing.energy) || 0))),
          pain: Math.max(0, Math.min(5, Math.round(Number(existing.pain) || 0))),
          libido: Math.max(0, Math.min(5, Math.round(Number(existing.libido) || 0))),
          stress: existing.stress !== undefined ? Math.max(0, Math.min(5, Math.round(Number(existing.stress) || 0))) : Math.max(0, Math.min(5, Math.round(legacyStress)))
        };
        emotionLevels = buildLoadedEmotionLevels(existing);
        tempIsPeriod = Boolean(existing.isPeriod || isCycleStart);
        tempIsOvulation = Boolean(existing.isOvulation);
        tempIntimacy = Boolean(existing.intimacy);
        tempSoloCount = normalizeSoloCount(existing);
        tempDischarge = tempIsPeriod
          ? (['Скудные', 'Умеренные', 'Обильные'].includes(existing.discharge) ? existing.discharge : 'Умеренные')
          : (['Сухие / скудные', 'Кремовые', 'Прозрачные', 'Мажущие'].includes(existing.discharge) ? existing.discharge : '');

        populateEditableInputs(existing);
      } else {
        metrics = { energy: 0, pain: 0, libido: 0, stress: 0 };
        emotionLevels = createDefaultEmotionLevels();
        tempIsPeriod = isCycleStart;
        tempIsOvulation = false;
        tempIntimacy = false;
        tempSoloCount = 0;
        tempDischarge = isCycleStart ? 'Умеренные' : '';

        populateEditableInputs({
          energy: 0, pain: 0, libido: 0, stress: 0,
          intimacy: false, soloCount: 0, soloReset: false, emotionLevels, contexts: [],
          wristTempDelta: ''
        });
      }

      updateTodaySaveStatus(false);
      updateHeaderAndContextControls();
      renderAdviceAndSelfCareTab();
    }

    function renderCalendarGrid() {
      const year = calendarViewDate.getFullYear();
      const month = calendarViewDate.getMonth();
      const title = new Date(year, month, 1).toLocaleDateString('ru-RU', { month: 'long', year: 'numeric' });
      document.getElementById('calMonthTitle').textContent = title;

      const firstDay = new Date(year, month, 1);
      let startWeekday = firstDay.getDay();
      startWeekday = startWeekday === 0 ? 6 : startWeekday - 1;
      const daysInMonth = new Date(year, month + 1, 0).getDate();
      const todayISO = getTodayISO(0);

      let html = '';
      for (let i = 0; i < startWeekday; i++) html += `<div class="cal-cell empty"></div>`;

      for (let day = 1; day <= daysInMonth; day++) {
        const cellDate = new Date(year, month, day, 12, 0, 0);
        const iso = formatLocalISO(cellDate);
        const info = getCycleInfo(iso);
        const entry = state.entries[iso];
        const isCycleStart = iso === state.settings.lastPeriodStart;

        const classes = ['cal-cell'];
        if (entry?.isPeriod || isCycleStart) classes.push('is-period');
        else if (info.isPredictedPeriod) classes.push('is-predicted-period');
        else if (entry?.isOvulation) classes.push('is-ovulation');
        else if (info.isPredictedOvulation) classes.push('is-predicted-ovulation');
        if (isCycleStart) classes.push('is-cycle-start');
        if (iso === todayISO) classes.push('is-today');
        if (iso === selectedDate) classes.push('is-selected');

        html += `
          <button class="${classes.join(' ')}" onclick="selectCalendarDate('${iso}')">
            <span>${day}</span>
            ${(entry || isCycleStart) ? '<span class="cal-dot"></span>' : ''}
          </button>
        `;
      }
      document.getElementById('calGrid').innerHTML = html;
    }

    function detectLocalPatterns(info) {
      const allEntries = Object.values(state.entries);
      if (allEntries.length < 4) return '';

      const samePhaseEntries = allEntries.filter(e => e.date !== selectedDate && getCycleInfo(e.date).phaseCode === info.phaseCode);
      const notes = [];

      if (samePhaseEntries.length >= 2) {
        const avgPain = (samePhaseEntries.reduce((s, e) => s + e.pain, 0) / samePhaseEntries.length).toFixed(1);
        const avgEnergy = (samePhaseEntries.reduce((s, e) => s + e.energy, 0) / samePhaseEntries.length).toFixed(1);
        const breastRate = samePhaseEntries.filter(e => (e.contexts || []).map(normalizeContextLabel).includes('Чувствительность груди')).length;
        notes.push(`В этой фазе у тебя обычно энергия ~${avgEnergy}/5, боль ~${avgPain}/5.`);
        if (breastRate >= 2) {
          notes.push(`Чувствительность груди повторялась в ${breastRate} из ${samePhaseEntries.length} дней этой фазы.`);
        }
      }

      const helpCounts = {};
      allEntries.forEach(e => (e.helped || []).forEach(h => { helpCounts[h] = (helpCounts[h] || 0) + 1; }));
      const topHelp = Object.entries(helpCounts).sort((a, b) => b[1] - a[1])[0];
      if (topHelp) {
        notes.push(`Чаще всего тебя выручало: «${topHelp[0]}» (${topHelp[1]} раз).`);
      }

      return notes.join(' ');
    }

          'Отметь самочувствие и выделения, чтобы видеть свой индивидуальный ритм.'

        'Сделай короткую паузу на дневной свет и движение в середине дня.',

    function toggleHelpedTip(tipId, isCustom = false) {
      hapticImpact('medium');
      const cleanId = String(tipId || '').trim();
      if (!cleanId) return;
      if (!state.entries[selectedDate]) {
        state.entries[selectedDate] = buildLiveDaySnapshot();
      }
      const list = Array.isArray(state.entries[selectedDate].helped) ? [...state.entries[selectedDate].helped] : [];
      const idx = list.indexOf(cleanId);
      if (idx >= 0) {
        list.splice(idx, 1);
        showToast('Отметка снята');
      } else {
        list.push(cleanId);
        if (isCustom) {
          if (!Array.isArray(state.customHelpedPool)) state.customHelpedPool = [];

      }

      state.entries[selectedDate].updatedAt = Date.now();
      persistLocalAndCloud(selectedDate);
      renderAdviceAndSelfCareTab();
    }

    function catmullRomPath(pts) {
      if (pts.length === 0) return '';
      if (pts.length === 1) return `M ${pts[0].x} ${pts[0].y}`;
      let d = `M ${pts[0].x.toFixed(1)} ${pts[0].y.toFixed(1)}`;
      for (let i = 0; i < pts.length - 1; i++) {
        const p0 = pts[i === 0 ? i : i - 1];
        const p1 = pts[i];
        const p2 = pts[i + 1];
        const p3 = p2.x !== undefined && pts[i + 2] ? pts[i + 2] : p2;
        const cp1x = p1.x + (p2.x - p0.x) / 6;
        const cp1y = p1.y + (p2.y - p0.y) / 6;
        const cp2x = p2.x - (p3.x - p1.x) / 6;
        const cp2y = p2.y - (p3.y - p1.y) / 6;
        d += ` C ${cp1x.toFixed(1)} ${cp1y.toFixed(1)}, ${cp2x.toFixed(1)} ${cp2y.toFixed(1)}, ${p2.x.toFixed(1)} ${p2.y.toFixed(1)}`;
      }
      return d;
    }

    function renderAnalytics() {
      const svg = document.getElementById('cycleChart');
      const scrollViewport = document.getElementById('chartScrollViewport');
      const baseLen = Number(state.settings.cycleLength) || 28;
      const pLen = Number(state.settings.periodLength) || 5;
      const isDark = Boolean(state.settings.darkMode);
      const colors = getCurrentPaletteColors();

      const cAccent = colors['--accent'];
      const cIndigo = colors['--indigo'];
      const cAmber = colors['--amber'];
      const cRose = colors['--rose-solid'];
      const cOvul = colors['--ovul'] || cAccent;

      const allSortedEntries = Object.values(state.entries).sort((a, b) => a.date.localeCompare(b.date));
      document.getElementById('chartEntriesCount').textContent = allSortedEntries.length
        ? pluralDays(allSortedEntries.length)
        : '';

      const byDay = {};
      let maxObservedDay = baseLen;
      let latestLoggedCycleDay = 1;
      allSortedEntries.forEach(e => {
        const d = Math.max(1, getCycleInfo(e.date).cycleDay);
        byDay[d] = e;
        if (d > maxObservedDay) maxObservedDay = d;
        latestLoggedCycleDay = d;
      });

      const len = Math.max(2, maxObservedDay);
      const effectivePadL = 18;
      const padR = 18;
      const padT = 44;
      const padB = 32;
      const h = 440;
      const plotH = h - padT - padB;
      const minDayStep = 26;
      const w = Math.max(400, effectivePadL + padR + (len - 1) * minDayStep);
      const effectivePlotW = w - effectivePadL - padR;

      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      svg.style.width = `${w}px`;
      svg.style.height = `${h}px`;

      const emotionScore = {};
      Object.values(byDay).forEach(e => {
        const lv = buildLoadedEmotionLevels(e);
        ALL_EMOTIONS.forEach(em => {
          if ((lv[em] ?? 0) > 0) {
            emotionScore[em] = (emotionScore[em] || 0) + lv[em];
          }
        });
      });
      const trackedEmotions = Object.entries(emotionScore)
        .sort((a, b) => b[1] - a[1])
        .map(([em]) => em);

      const subRow = document.getElementById('emotionSubFilters');
      if (chartFilter === 'emotions') {
        subRow.style.display = 'flex';
        const filterList = trackedEmotions.length ? trackedEmotions : ALL_EMOTIONS.slice(0, 6);
        if (selectedEmotionChartFilter !== 'all' && !filterList.includes(selectedEmotionChartFilter)) {
          selectedEmotionChartFilter = 'all';
        }
        const pills = ['all', ...filterList];
        subRow.innerHTML = pills.map(em => {
          const active = selectedEmotionChartFilter === em;
          const label = em === 'all' ? 'Все эмоции' : em;
          return `<button class="filter-pill ${active ? 'active' : ''}" onclick="setEmotionChartFilter('${em.replace(/'/g, "\'")}')">${escapeHtml(label)}</button>`;
        }).join('');
      } else {
        subRow.style.display = 'none';
      }

      const dayToX = (d) => effectivePadL + ((d - 1) / (len - 1)) * effectivePlotW;

      let html = `<defs>
        <linearGradient id="gradAccent" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stop-color="${cAccent}" stop-opacity="0.22"/>
          <stop offset="100%" stop-color="${cAccent}" stop-opacity="0"/>
        </linearGradient>
        <linearGradient id="gradIndigo" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stop-color="${cIndigo}" stop-opacity="0.20"/>
          <stop offset="100%" stop-color="${cIndigo}" stop-opacity="0"/>
        </linearGradient>
        <linearGradient id="gradAmber" x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stop-color="${cAmber}" stop-opacity="0.20"/>
          <stop offset="100%" stop-color="${cAmber}" stop-opacity="0"/>
        </linearGradient>
      </defs>`;

      const wMens = ((Math.min(pLen, len) - 1) / (len - 1)) * effectivePlotW;
      html += `<rect x="${effectivePadL}" y="${padT}" width="${Math.max(12, wMens)}" height="${plotH}" rx="8" fill="${isDark ? 'rgba(242,107,133,0.12)' : 'rgba(217,72,101,0.08)'}" />`;

      const pmsStart = Math.max(1, baseLen - 6);
      if (pmsStart < len) {
        const xPms = dayToX(pmsStart);
        const wPms = ((Math.min(baseLen, len) - pmsStart) / (len - 1)) * effectivePlotW;
        if (wPms > 0) {
          html += `<rect x="${xPms}" y="${padT}" width="${wPms}" height="${plotH}" rx="8" fill="${isDark ? 'rgba(232,162,84,0.10)' : 'rgba(201,122,43,0.08)'}" />`;
        }
      }

      // Horizontal reference levels (0..5)
      [0, 1, 2, 3, 4, 5].forEach(v => {
        const y = padT + plotH - (v / 5) * plotH;
        const isMajor = v === 0 || v === 5;
        html += `<line x1="${effectivePadL}" y1="${y}" x2="${w - padR}" y2="${y}" stroke="var(--border)" stroke-opacity="${isMajor ? '0.95' : '0.55'}" stroke-dasharray="${isMajor ? 'none' : '3 3'}" />`;
      });

      // Day ticks along bottom axis for orientation when scrolling
      const dayStepLabel = w > 420 ? 2 : 4;
      for (let d = 1; d <= len; d++) {
        const x = dayToX(d);
        if (byDay[d]) {
          html += `<line x1="${x.toFixed(1)}" y1="${padT}" x2="${x.toFixed(1)}" y2="${padT + plotH}" stroke="var(--border)" stroke-opacity="0.32" />`;
        }
        if (d === 1 || d === len || d % dayStepLabel === 0) {
          html += `<text x="${x.toFixed(1)}" y="${h - 9}" text-anchor="middle" font-size="11" font-weight="600" fill="var(--ink-3)">${d}</text>`;
        }
      }

      // Smart separation & layered stroke weights so coinciding curves remain clearly visible
      function renderSeriesCollection(seriesList) {
        const bucketMap = {};
        seriesList.forEach((s, sIdx) => {
          s.pts.forEach((p, pIdx) => {
            const valKey = typeof p.normKey === 'number' ? p.normKey : Math.round(p.val * 10);
            const key = `${p.day}_${valKey}`;
            if (!bucketMap[key]) bucketMap[key] = [];
            bucketMap[key].push({ sIdx, pIdx, p });
          });
        });

        Object.values(bucketMap).forEach(group => {
          const count = group.length;
          group.forEach((item, idx) => {
            item.p.overlapCount = count;
            item.p.overlapRank = idx;
            if (count > 1) {
              const step = count === 2 ? 10.5 : (count === 3 ? 9.5 : 8.0);
              const offset = (idx - (count - 1) / 2) * step;
              item.p.y = Math.max(padT + 8, Math.min(padT + plotH - 8, item.p.baseY + offset));
            } else {
              item.p.y = item.p.baseY;
            }
          });
        });

        // 1. Draw area gradients first (if any)
        seriesList.forEach(s => {
          if (s.pts.length > 1 && s.gradId) {
            const pathD = catmullRomPath(s.pts);
            const first = s.pts[0];
            const last = s.pts[s.pts.length - 1];
            const baseY = padT + plotH;
            html += `<path d="${pathD} L ${last.x.toFixed(1)} ${baseY} L ${first.x.toFixed(1)} ${baseY} Z" fill="url(#${s.gradId})" />`;
          }
        });

        // 2. Draw curves with distinct widths & opacity
        seriesList.forEach(s => {
          if (!s.pts.length) return;
          const sw = s.strokeWidth || 2.8;
          const op = s.opacity !== undefined ? s.opacity : 0.95;
          const dash = s.dashArray ? ` stroke-dasharray="${s.dashArray}"` : '';
          if (s.pts.length > 1) {
            const pathD = catmullRomPath(s.pts);
            html += `<path d="${pathD}" fill="none" stroke="${s.color}" stroke-width="${sw}" stroke-opacity="${op}"${dash} stroke-linecap="round" stroke-linejoin="round" />`;
          } else {
            const p = s.pts[0];
            html += `<line x1="${p.x}" y1="${padT}" x2="${p.x}" y2="${padT + plotH}" stroke="${s.color}" stroke-opacity="0.25" stroke-dasharray="2 2" />`;
            html += `<circle cx="${p.x}" cy="${p.y}" r="9" fill="${s.color}" opacity="0.20" />`;
          }
        });

        // 3. Draw clean points without outer concentric rings
        seriesList.forEach(s => {
          s.pts.forEach(p => {
            const dotR = s.dotRadius || 4.4;
            html += `<circle cx="${p.x.toFixed(1)}" cy="${p.y.toFixed(1)}" r="${dotR}" fill="${s.color}" stroke="var(--surface)" stroke-width="1.6" />`;
          });
        });
      }

      const tempPtsRaw = [];
      for (let d = 1; d <= len; d++) {
        const e = byDay[d];
        if (e && typeof e.wristTempDelta === 'number' && !isNaN(e.wristTempDelta)) {
          tempPtsRaw.push({ day: d, val: e.wristTempDelta });
        }
      }

      const legendEl = document.getElementById('chartLegendRow');

      if (chartFilter === 'temp') {
        if (tempPtsRaw.length) {
          const vals = tempPtsRaw.map(p => p.val);
          const rawMin = Math.min(...vals);
          const rawMax = Math.max(...vals);
          const minT = rawMin === rawMax ? rawMin - 0.5 : rawMin - Math.max(0.15, (rawMax - rawMin) * 0.25);
          const maxT = rawMin === rawMax ? rawMax + 0.5 : rawMax + Math.max(0.15, (rawMax - rawMin) * 0.25);
          const pts = tempPtsRaw.map(p => {
            const ratio = (p.val - minT) / Math.max(0.01, maxT - minT);
            const yPos = padT + plotH - Math.max(0, Math.min(1, ratio)) * plotH;
            return {
              day: p.day,
              x: dayToX(p.day),
              baseY: yPos,
              y: yPos,
              val: p.val
            };
          });
          renderSeriesCollection([{
            pts,
            color: cAccent,
            gradId: 'gradAccent',
            strokeWidth: 3.1,
            dotRadius: 4.8
          }]);
        }
        legendEl.innerHTML = `<span class="legend-item" style="color:${cAccent};">${SVG_ICONS.dot}<span>Температура (°C)</span></span>`;
      } else if (chartFilter === 'emotions') {
        const emosToDraw = selectedEmotionChartFilter === 'all'
          ? (trackedEmotions.length ? trackedEmotions.slice(0, 5) : [])
          : [selectedEmotionChartFilter];

        const emoWidths = [3.8, 3.1, 2.5, 2.1, 1.8];
        const seriesList = emosToDraw.map((emo, idx) => {
          const pal = EMOTION_PALETTE[emo] || { color: cAccent };
          const pts = [];
          for (let d = 1; d <= len; d++) {
            const e = byDay[d];
            if (e) {
              const lv = buildLoadedEmotionLevels(e);
              const val = Math.min(5, lv[emo] ?? 0);
              if (val > 0 || selectedEmotionChartFilter !== 'all') {
                const yPos = padT + plotH - (val / 5) * plotH;
                pts.push({
                  day: d,
                  x: dayToX(d),
                  baseY: yPos,
                  y: yPos,
                  val,
                  normKey: val
                });
              }
            }
          }
          return {
            pts,
            color: pal.color,
            gradId: null,
            strokeWidth: emoWidths[idx] || 2.4,
            opacity: 0.92,
            dotRadius: Math.max(3.8, 4.8 - idx * 0.25)
          };
        });

        renderSeriesCollection(seriesList);

        legendEl.innerHTML = emosToDraw.length
          ? emosToDraw.map(em => {
              const c = (EMOTION_PALETTE[em] || { color: cAccent }).color;
              return `<span class="legend-item" style="color:${c};">${SVG_ICONS.dot}<span>${escapeHtml(em)}</span></span>`;
            }).join('')
          : '';
      } else {
        const buildMetricPts = (key) => {
          const pts = [];
          for (let d = 1; d <= len; d++) {
            if (byDay[d] && typeof byDay[d][key] === 'number') {
              const val = Math.max(0, Math.min(5, byDay[d][key]));
              const yPos = padT + plotH - (val / 5) * plotH;
              pts.push({
                day: d,
                x: dayToX(d),
                baseY: yPos,
                y: yPos,
                val,
                normKey: val
              });
            }
          }
          return pts;
        };

        const metricSeries = [];
        if (chartFilter === 'all' || chartFilter === 'energy') {
          metricSeries.push({
            pts: buildMetricPts('energy'),
            color: cIndigo,
            gradId: chartFilter === 'energy' ? 'gradIndigo' : null,
            strokeWidth: 3.8,
            opacity: 0.86,
            dotRadius: 4.8
          });
        }
        if (chartFilter === 'all' || chartFilter === 'libido') {
          metricSeries.push({
            pts: buildMetricPts('libido'),
            color: cAmber,
            gradId: chartFilter === 'libido' ? 'gradAmber' : null,

            opacity: 0.94,

            pts: buildMetricPts('pain'),
            color: cRose,
            gradId: null,
            strokeWidth: 2.1,
            opacity: 0.98,
            dotRadius: 4.0
          });

        renderSeriesCollection(metricSeries);

        legendEl.innerHTML = `
          <span class="legend-item" style="color:${cIndigo};">${SVG_ICONS.dot}<span>Энергия</span></span>
          <span class="legend-item" style="color:${cAmber};">${SVG_ICONS.dot}<span>Либидо</span></span>
          <span class="legend-item" style="color:${cRose};">${SVG_ICONS.dot}<span>Боль</span></span>
        `;
      }

      for (let d = 1; d <= len; d++) {
        const e = byDay[d];
        if (!e) continue;
        const x = dayToX(d);
        const soloCnt = normalizeSoloCount(e);
        const topMarkers = [];
        if (e.intimacy) topMarkers.push({ type: 'heart', color: 'var(--rose)' });
        if (soloCnt > 0) topMarkers.push({ type: 'bolt', color: cAmber });
        if (e.isOvulation) topMarkers.push({ type: 'sparkle', color: cOvul });

        if (topMarkers.length) {
          const spacing = 16;
          const startX = x - ((topMarkers.length - 1) * spacing) / 2;
          topMarkers.forEach((m, idx) => {
            const mx = startX + idx * spacing;
            const my = padT - 24;
            if (m.type === 'heart') {
              html += `<g transform="translate(${(mx - 6.5).toFixed(1)}, ${my})" fill="${m.color}"><path d="M6.5 11.6L5.7 10.9C2.9 8.3 1.1 6.7 1.1 4.6 1.1 2.9 2.4 1.6 4.1 1.6c1 0 1.9.4 2.4 1.1.5-.7 1.4-1.1 2.4-1.1 1.7 0 3 1.3 3 3 0 2.1-1.8 3.7-4.6 6.3l-.8.7z"/></g>`;
            } else if (m.type === 'bolt') {
              html += `<g transform="translate(${(mx - 6.5).toFixed(1)}, ${my})" fill="${m.color}"><polygon points="7.2 1 1.6 7.6 6.6 7.6 6.1 12 11.6 5.4 6.6 5.4 7.2 1"/></g>`;
            } else if (m.type === 'sparkle') {
              html += `<g transform="translate(${(mx - 6.5).toFixed(1)}, ${my})" fill="${m.color}"><path d="M6.5 1l1.3 4.2L12 6.5 7.8 7.8 6.5 12 5.2 7.8 1 6.5l4.2-1.3L6.5 1z"/></g>`;
            }
          });
        }
      }

      svg.innerHTML = html;

      if (scrollViewport && w > scrollViewport.clientWidth && scrollViewport.clientWidth > 0) {
        const targetX = dayToX(latestLoggedCycleDay);
        scrollViewport.scrollLeft = Math.max(0, targetX - scrollViewport.clientWidth * 0.65);
      }
    }

    function deletePainkiller(index) {
      hapticImpact('light');

        persistLocalAndCloud('__meta_only__');
        updatePainkillerUI();
        showToast('Удалено');

    function updatePainkillerUI() {
      const box = document.getElementById('painkillerStatus');
      if (!state.painkillers.length) {
        box.innerHTML = '';
        return;
      }
      box.innerHTML = state.painkillers.slice(0, 12).map((item, idx) => {
        const taken = new Date(item.timeISO);
        const nextSafe = new Date(taken.getTime() + (item.hours || 4) * 3600000);
        const dayStr = taken.toLocaleDateString('ru-RU', { day: 'numeric', month: 'short' }).replace('.', '');
        const tStr = taken.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
        const nStr = nextSafe.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
        return `
          <div class="med-item">
            <div class="med-item-info">
              <div class="med-item-name">${escapeHtml(item.name)}</div>
              <div class="med-item-time">${dayStr}, ${tStr} → след. с ${nStr}</div>
            </div>
            <button type="button" class="med-del-btn" onclick="deletePainkiller(${idx})" aria-label="Удалить">${SVG_ICONS.cross}</button>
          </div>
        `;
      }).join('');

    function selectBreathPattern(patternKey) {
      if (!BREATH_PATTERNS[patternKey]) return;
      hapticSelect();
      const wasRunning = isBreathing;
      if (isBreathing) stopBreathing();
      currentBreathPattern = patternKey;
      renderBreathPatternUI();
      if (wasRunning) toggleBreathing();
    }

    function stopBreathing() {
      isBreathing = false;
      if (breathTimer) { clearTimeout(breathTimer); breathTimer = null; }
      if (breathTickTimer) { clearInterval(breathTickTimer); breathTickTimer = null; }
      const motion = document.getElementById('flowerMotion');
      const prompt = document.getElementById('flowerPrompt');
      const sub = document.getElementById('flowerSubPrompt');
      const btn = document.getElementById('breathBtn');
      motion.style.transitionDuration = '0.6s';
      motion.className = 'flower-motion';
      prompt.textContent = '';
      sub.textContent = '';
      btn.textContent = 'Начать';
    }

    function toggleBreathing() {
      hapticImpact('medium');
      if (isBreathing) { stopBreathing(); return; }

      const pat = BREATH_PATTERNS[currentBreathPattern] || BREATH_PATTERNS.spasm;
      isBreathing = true;
      document.getElementById('breathBtn').textContent = 'Стоп';
      const motion = document.getElementById('flowerMotion');
      const prompt = document.getElementById('flowerPrompt');
      const sub = document.getElementById('flowerSubPrompt');

      let stepIndex = 0;

      const runStep = () => {
        if (!isBreathing) return;
        const step = pat.steps[stepIndex];
        motion.style.transitionDuration = `${(step.ms / 1000).toFixed(2)}s`;
        motion.className = `flower-motion ${step.cls}`;
        prompt.textContent = step.label;

        if (step.haptic === 'medium') hapticImpact('medium');
        else if (step.haptic === 'select') hapticSelect();
        else hapticImpact('light');

        if (breathTickTimer) clearInterval(breathTickTimer);
        let remainingSec = Math.ceil(step.ms / 1000);
        sub.textContent = `${remainingSec} с`;
        breathTickTimer = setInterval(() => {
          remainingSec -= 1;
          if (remainingSec > 0 && isBreathing) {
            sub.textContent = `${remainingSec} с`;
          }
        }, 1000);

        breathTimer = setTimeout(() => {
          stepIndex = (stepIndex + 1) % pat.steps.length;
          runStep();
        }, step.ms);
      };

      runStep();
    }

    function saveSettings() {
      const lp = document.getElementById('setLastPeriod').value;
      const cl = Number(document.getElementById('setCycleLength').value) || 28;
      const pl = Number(document.getElementById('setPeriodLength').value) || 5;

      if (lp && lp !== state.settings.lastPeriodStart) {
        state.settings.prevLastPeriodStart = state.settings.lastPeriodStart;
        state.settings.lastPeriodStart = lp;
      }
      state.settings.cycleLength = Math.max(21, Math.min(45, cl));
      state.settings.periodLength = Math.max(2, Math.min(10, pl));
      state.onboarded = true;

      syncSettingsInputs();
      persistLocalAndCloud('__meta_only__');
      loadDayForm();
      renderAnalytics();
      closeSheet('sheetCycleSettings');
      showToast('Сохранено');
    }
